import ollama
def generate_fun(d):
    promp = f"""
     Write a python function based on the description{d}
     Return python code only
     """
    response= ollama.chat(
         model='phi:latest',
         messages=[{
             'role':'user',
             'content':promp
         }]
             
         )
    return response ['message']['content']

def validate(code,testcase):
    r={}
    exec(code,r)
    function= r['add_number']
    for input,expect in testcase:
        output=function(**input)
        if(output==expect):
            print("pass")
        
        else:
            print("fail")
        
        
def main():
    d="Write a python function named add_number that returns sum of two numbers "
    code=generate_fun(d)
    print(code)
    testcase=[
        ({'a':2,'b':3},5),
        ({'a':1,'b':3},4),
        ({'a':2,'b':1},3),
        ({'a':5,'b':5},10)
        
    ]
    validate(code,testcase)
main()
